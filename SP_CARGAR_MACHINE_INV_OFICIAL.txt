           OR src.ID_UBICACION IS NULL
           OR src.ID_SERVICIOS IS NULL
           OR src.ID_CLUSTER_VM IS NULL
         THEN 'ERROR' ELSE 'OK' END,
    TRIM(
      CASE WHEN src.ID_SISTEMA_OPERATIVO IS NULL THEN 'No se encontro SISTEMA_OPERATIVO en catalogo. ' END ||
      CASE WHEN src.ID_AMBIENTE_INV IS NULL THEN 'No se encontro AMBIENTE en catalogo. ' END ||
      CASE WHEN src.ID_UBICACION IS NULL THEN 'No se encontro UBICACION en catalogo. ' END ||
      CASE WHEN src.ID_SERVICIOS IS NULL THEN 'No se encontro SERVICIOS en catalogo. ' END ||
      CASE WHEN src.ID_CLUSTER_VM IS NULL THEN 'No se encontro CLUSTER_VM en catalogo. ' END
    )
  );

  COMMIT;

EXCEPTION
  WHEN OTHERS THEN
    ROLLBACK;
    RAISE;
END SP_CARGAR_MACHINE_INV;
